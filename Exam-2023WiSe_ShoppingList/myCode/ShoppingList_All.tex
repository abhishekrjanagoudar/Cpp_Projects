float Product::priceFor(float quantity) const {
	float totalPrice = basePrice * quantity;
	float discountValue = discount->discountFor(quantity);
	return totalPrice * (1 - discountValue);
}

Shop& Shop::setBasePrice(std::string productName, float basePrice) {
	auto it = products.find(productName);
	if (it == products.end()) {
		throw std::invalid_argument("Error");
	}
	it->second.setBasePrice(basePrice);
	return *this;
}
//setDiscount Same assetBasePrice
float Shop::calculatePurchase(const ShoppingList& shoppingList,
		std::set<const Item*>& notAvailable) const {
    float totalPrice = 0;
    list<Item>::const_iterator begin;
    shoppingList.items(begin, end);
    for (auto it = begin; it != end; ++it) {
        auto productIt = products.find(it->getName());
        if (productIt != products.end()) {
            totalPrice += productIt->second.priceFor(it->getQuantity());
        } else {
            notAvailable.insert(&(*it));}}
    return totalPrice;}
\end{lstlisting}

\section*{ShopDb.cpp}

using namespace std;
ShopDb& ShopDb::addShop(std::unique_ptr<Shop>&& shop) {
	knownShops[shop->getName()] = std::move(shop);
	return *this;
}
vector<Shop*> ShopDb::shops() const {
    vector<Shop*> result;
    for (auto& pair : knownShops) {
        result.push_back(pair.second.get());
    }
    return result;
}
Shop* ShopDb::shopByName(std::string name) {
	auto it = knownShops.find(name);
	if (it != knownShops.end()){
		return it->second.get();
	}
	return nullptr;
}
