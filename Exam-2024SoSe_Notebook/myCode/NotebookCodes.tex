void Note::printItem(ostream& out, int indent) const {
    Item::printItem(out, indent);
    out << string(indent + 3, ' ')
        << content
        << endl;
}
void Note::writeCsv(std::ostream& out, const Item* parent) const {
    out << "Note;"
        << getId() << ";"
        << (parent ? parent->getId() : 0) << ";"
        << getTitle() << ";"
        << content
        << endl;
}
bool Note::fromCsv(std::map<long, Topic*>& topics, std::string line) {
    stringstream ss(line);
    string type;
    string idStr;
    string parentIdStr;
    string title;
    string content;
    getline(ss, type, ';');
    if (type != "Note") {
        return false;
    }
    getline(ss, idStr, ';');
    getline(ss, parentIdStr, ';');
    getline(ss, title, ';');
    getline(ss, content);
    long parentId = stol(parentIdStr);
    Note* note = new Note(title);
    note->setContent(content);
    auto parent = topics[parentId];
    parent->add(unique_ptr<Item>(note));
    return true;
}

\section*{Topic.cpp}
\begin{lstlisting}
void Topic::add(std::unique_ptr<Item>& item) {
    items.push_back(std::move(item));
}
void Topic::add(std::unique_ptr<Item>&& item) {
    items.push_back(std::move(item));
}
void Topic::printItem(ostream& out, int indent) const {
    Item::printItem(out, indent);
    for (const auto& item : items) {
        item->printItem(out, indent + 3);
    }
}
void Topic::writeCsv(std::ostream& out, const Item* parent) const {
    out << "Topic" << ";"
        << getId() << ";"
        << (parent ? parent->getId() : 0) << ";"
        << getTitle() << ";"
        << endl;
    for (const auto& item : items) {
        item->writeCsv(out, this);
    }
}
bool Topic::fromCsv(std::map<long, Topic*>& topics, std::string line) {
    stringstream ss(line);
    string type;
    string idStr;
    string parentIdStr;
    string title;
    getline(ss, type, ';');
    if (type != "Topic") {
        return false;
    }
    getline(ss, idStr, ';');
    getline(ss, parentIdStr, ';');
    getline(ss, title, ';');
    long id = stol(idStr);
    long parentId = stol(parentIdStr);
    Topic* topic = new Topic(title);
    topics[id] = topic;
    if (parentId != 0) {
        auto parent = topics[parentId];
        parent->add(unique_ptr<Item>(topic));
    }
    return true;
}
\end{lstlisting}

\section*{Notebook.cpp}
\begin{lstlisting}
void Notebook::print(ostream& out) const {
    string caption = "Notebook: " + getTitle();
    out << caption << endl;
    out << string(caption.length(), '-') << endl;
    for (auto item : getItems()) {
        item->printItem(out, 3);
    }
}
void Notebook::print() const {
    print(cout);
}
void Notebook::saveAsCsv(std::ostream& out) const {
    writeCsv(out, nullptr);
}
Notebook* Notebook::loadFromCsv(std::istream& in) {
    map<long, Topic*> topics;
    string line;
    while (getline(in, line)) {
        if (Topic::fromCsv(topics, line)) {
            continue;
        }
        Note::fromCsv(topics, line);
    }
    if (topics.empty()) {
        return nullptr;
    }
    Topic* rootTopic = topics.begin()->second;
    Notebook* notebook =
            new Notebook(rootTopic->getTitle());
    delete rootTopic;
    topics.begin()->second = notebook;
    return notebook;
}

void assertTrue(bool condition, string failedMessage) {
    if (!condition) {
        cout << failedMessage << endl;
    }
}
void noteTests() {
    Note note("Test Note");
    note.setContent("Lorem ipsum");
    assertTrue(note.getTitle() == "Test Note",
            "ERROR: Note title");
    assertTrue(note.getContent() == "Lorem ipsum",
            "ERROR: Note content");
    stringstream ss;
    note.printItem(ss, 7);
    string expected =
            "       - Test Note\n"
            "          Lorem ipsum\n";
    assertTrue(ss.str() == expected,
            "ERROR: Note print");
    Note note2("Another Note");
    assertTrue(note2.getId() == note.getId() + 1,
            "ERROR: Note id");
}
void topicTests() {
    Topic topic("Test Topic");
    assertTrue(topic.getTitle() == "Test Topic",
            "ERROR: Topic title");
    assertTrue(topic.getItems().size() == 0,
            "ERROR: Topic should be empty");
    stringstream ss;
    topic.writeCsv(ss, nullptr);
    string expectedTopic =
            "Topic;" +
            to_string(topic.getId()) +
            ";0;Test Topic;\n";
    assertTrue(ss.str() == expectedTopic,
            "ERROR: Topic CSV");
    auto note =
        unique_ptr<Item>(
            (new Note("Test Note"))
            ->setContent("Lorem ipsum"));
    Item* addedNote = note.get();
    topic.add(std::move(note));
    assertTrue(topic.getItems().size() == 1,
            "ERROR: Topic item count");
    assertTrue(topic.getItems()[0] == addedNote,
            "ERROR: Wrong item stored");
    stringstream ss2;
    addedNote->writeCsv(ss2, &topic);
    string expectedNote =
            "Note;" +
            to_string(addedNote->getId()) +
            ";" +
            to_string(topic.getId()) +
            ";Test Note;Lorem ipsum\n";
    assertTrue(ss2.str() == expectedNote,
            "ERROR: Note CSV");
}
void notebookTests() {
    Notebook *notebook = new Notebook("Studies Notebook");
    notebook->add(std::unique_ptr<Item>((new Note("Work consistently"))
        ->setContent("Plan ahead!")));
    Topic* apt = new Topic("APT");
    apt->add(std::unique_ptr<Item>((new Note("Learn C++"))
        ->setContent("Remember that this takes some time!")));
    apt->add(std::unique_ptr<Item>(new Note("Learn about libraries")));
    notebook->add(std::unique_ptr<Item>(apt));
    Topic* other = new Topic("Other");
    other->add(std::unique_ptr<Item>(new Note("Learn Something")));
    other->add(std::unique_ptr<Item>(new Note("Learn more about something")));
    notebook->add(std::unique_ptr<Item>(other));
    notebook->add(std::unique_ptr<Item>(new Note("Finish the studies")));
    ofstream out("notebook.csv");
    notebook->saveAsCsv(out);
    out.close();
    ifstream in("notebook.csv");
    Notebook* restored =
            Notebook::loadFromCsv(in);
    in.close();
    stringstream originalText;
    stringstream restoredText;
    notebook->print(originalText);
    restored->print(restoredText);
    assertTrue(
        originalText.str() == restoredText.str(),
        "ERROR: Notebook restore"
    );
    delete notebook;
    delete restored;
}
void allTests() {
    noteTests();
    topicTests();
    notebookTests();
}
